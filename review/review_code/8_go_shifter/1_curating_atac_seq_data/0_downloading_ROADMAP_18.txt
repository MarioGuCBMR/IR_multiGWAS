#!/bin/bash
# =============================================================================
# Download all 98 Roadmap 18-state chromatin state segmentations (mnemonics BED)
# 18-state model = core 5 marks + H3K27ac, learned on 98 epigenomes
# Source: https://egg2.wustl.edu/roadmap/data/byFileType/chromhmmSegmentations/
#         ChmmModels/core_K27ac/jointModel/final/
# Format: BED (chr, start, end, state_mnemonic), gzipped, hg19
# EID list verified from live directory listing (98 files confirmed)
# =============================================================================

set -euo pipefail

OUTDIR="/projects/kilpelainen-AUDIT/people/zlc436/IR_GSEM_2025/review/review_raw_data/roadmap"
mkdir -p "$OUTDIR"

BASE="https://egg2.wustl.edu/roadmap/data/byFileType/chromhmmSegmentations/ChmmModels/core_K27ac/jointModel/final"

# 98 epigenome IDs with H3K27ac data — verified from directory listing
EIDS=(
  E003 E004 E005 E006 E007 E008 E011 E012 E013 E014
  E015 E016 E017 E019 E020 E021 E022 E026 E029 E032
  E034 E037 E038 E039 E040 E041 E042 E043 E044 E045
  E046 E047 E048 E049 E050 E055 E056 E058 E059 E061
  E062 E063 E065 E066 E067 E068 E069 E071 E072 E073
  E074 E075 E076 E078 E079 E080 E084 E085 E087 E089
  E090 E091 E092 E093 E094 E095 E096 E097 E098 E099
  E100 E101 E102 E103 E104 E105 E106 E108 E109 E111
  E112 E113 E114 E115 E116 E117 E118 E119 E120 E121
  E122 E123 E124 E125 E126 E127 E128 E129
)

echo "Downloading ${#EIDS[@]} epigenome 18-state segmentation files to ${OUTDIR}/"
echo ""

success=0
fail=0

for EID in "${EIDS[@]}"; do
  FILE="${EID}_18_core_K27ac_mnemonics.bed.gz"
  if [ -f "${OUTDIR}/${FILE}" ] && [ -s "${OUTDIR}/${FILE}" ]; then
    echo "  [SKIP] ${FILE} already exists"
    success=$((success + 1))
    continue
  fi
  curl -sL "${BASE}/${FILE}" -o "${OUTDIR}/${FILE}"
  if [ -s "${OUTDIR}/${FILE}" ]; then
    echo "  [OK]   ${FILE} ($(ls -lh ${OUTDIR}/${FILE} | awk '{print $5}'))"
    success=$((success + 1))
  else
    echo "  [FAIL] ${FILE}"
    rm -f "${OUTDIR}/${FILE}"
    fail=$((fail + 1))
  fi
done

# Also download the epigenome metadata table
META_FILE="${OUTDIR}/EID_metadata.tab"
if [ ! -f "${META_FILE}" ] || [ ! -s "${META_FILE}" ]; then
  echo ""
  echo "Downloading epigenome metadata..."
  curl -sL "https://egg2.wustl.edu/roadmap/data/byFileType/metadata/EID_metadata.tab" -o "$META_FILE"
fi
